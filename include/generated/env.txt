#include <env/ti/ti_common.env>
#include <env/ti/mmc.env>
#include <env/ti/k3_dfu.env>

/*
 * Since 0x80200000 is the load address for OPTEE, in order to avoid
 * potentially overwriting it with u-boot.img which has the default
 * load address of 0x81000000 as specified by dfu_alt_info_ram,
 * change the load address of u-boot.img to 0x83000000.
 */
#if CONFIG_TI_GPMC
mtdids=nand0=nand0
mtdparts=nand0:512k(NAND.tiboot3),1536k(NAND.tispl),2m(NAND.u-boot),512k(NAND.tiboot3.backup),1536k(NAND.tispl.backup),2m(NAND.u-boot.backup),256k(NAND.u-boot-env),256k(NAND.u-boot-env.backup),-(NAND.file-system)

dfu_alt_info_nand=
	NAND.tiboot3 part 0 1;
	NAND.tispl part 0 2;
	NAND.u-boot part 0 3;
	NAND.tiboot3.backup part 0 4;
	NAND.tispl.backup part 0 5;
	NAND.u-boot.backup part 0 6;
	NAND.u-boot-env part 0 7;
	NAND.u-boot-env.backup part 0 8;
	NAND.file-system part 0 9
#endif

dfu_alt_info_ram=
	tispl.bin ram 0x80080000 0x200000;
	u-boot.img ram 0x83000000 0x400000;
	script ram 0x82000000 0x40000

/* stage2 — typed first: erase eMMC + arm eMMC-DFU mode */
dfu-init=mmc dev 0 1; mmc erase 0 0xfc00; mmc dev 0 0; mmc erase 0 0x1da4000; gpt write mmc 0 ${partitions}; setenv dfu_alt_info ${dfu_alt_info_emmc}; dfu 0 mmc 0


/* stage4 — typed second: partconf + bootbus config */
dfu-flash=mmc dev 0 0; mmc partconf 0 1 1 1; mmc bootbus 0 2 0 0

/* internal only — never typed manually, both macros chain into this */

/*
 * Network flasher — mirrors dfu-init/dfu-flash pattern above.
 * Typed manually at U-Boot prompt, not wired into default bootcmd,
 * to avoid accidental netboot on normal power-up.
 */
flasher-fetch=dhcp; tftpboot ${loadaddr} Image && tftpboot ${fdtaddr} ${fdtfile}
flasher-boot=booti ${loadaddr} - ${fdtaddr}
bootcmd_flasher=run flasher-fetch && run flasher-boot


name_kern=Image
console=ttyS0,115200n8
args_all=setenv optargs ${optargs} earlycon=ns16550a,mmio32,0x02800000
	${mtdparts}
run_kern=booti ${loadaddr} ${rd_spec} ${fdtaddr}
boot_targets=mmc0 mmc1
boot=mmc
mmcdev=0
bootpart=0:1
bootdir=/boot
rd_spec=-
fdtaddr=0x84000000
